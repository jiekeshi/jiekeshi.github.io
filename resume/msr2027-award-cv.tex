\section{\textbf{\uppercase{Research Overview}}}

\cvitem{}{Software Engineering, Program Synthesis, LLM4Code, Performance Engineering, Software Analytics, Mining Software Repositories, Empirical Study, Software Testing}
\vspace{0.5em}

\cvitem{}{I develop \textbf{synthesis-assisted techniques} to \textbf{optimize the performance} of AI software, particularly LLMs trained on code from open-source repositories, and to \textbf{validate the safety} of software systems, from finding bugs in \textbf{core software infrastructure} to detecting safety violations in \textbf{AI systems that use neural controllers}.}
\vspace{0.5em}

\cvitem{}{For the MSR community, my work supports the efficient deployment of models learned from open-source code repositories and the validation of software reliability using mined knowledge and synthesized test artifacts.}
\vspace{0.5em}

\cvitem{}{I have published \textbf{43 papers} in top-tier venues such as ICSE, ASE, FSE, MSR, TOSEM, and TSE, \textbf{16 of which list me as first, co-first, or corresponding author}. My work has received \textbf{7 paper awards and honors} and \textbf{2 reviewer/PC awards}. My research has also had \textbf{real-world open-source impact}, including reporting \textbf{504 bugs, vulnerabilities, and license violations} in projects such as GCC, Clang, rustc, OpenSSL, curl, FFmpeg, and OP-TEE, of which \textbf{246 have been confirmed or fixed}.}
\vspace{0.2em}
